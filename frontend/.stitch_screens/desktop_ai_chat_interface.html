#!/usr/bin/env bash
set -euo pipefail

NAMESPACE="production"
DEPLOYMENT="auth-microservice"

# 1. Immediate Pod Memory Spike Inspection
echo "=== Pod Top Memory Usage ==="
kubectl top pods -n "$NAMESPACE" -l app="$DEPLOYMENT" --sort-by=memory

# 2. Check for recent OOMKilled events and exit code 137
echo "=== Inspecting Exit Codes and Restarts ==="
kubectl get pods -n "$NAMESPACE" -l app="$DEPLOYMENT" \
  -o jsonpath='{range .items[*]}{.metadata.name}{"\tRestarts: "}{.status.containerStatuses[0].restartCount}{"\tLastState: "}{.status.containerStatuses[0].lastState.terminated.reason}{"\n"}{end}'

# 3. Stream Live 504 NGINX Ingress Error Rate
echo "=== Aggregating 5xx Ingress Upstream Timeouts ==="
kubectl logs -n ingress-nginx -l app.kubernetes.io/name=ingress-nginx --tail=200 \
  | grep " 504 " | awk '{print $1, $4, $7, $9}' | head -n 10
